Kubernetes: comandos en orden
Workshop 1: Imperativo (sin YAML)
1. Entornokubectl version --clientkubectl config current-contextkubectl cluster-infokubectl get nodes2. Namespacekubectl create namespace curso-k8skubectl get namespace curso-k8skubectl get pods -n curso-k8sSi ya existe:kubectl delete namespace curso-k8s --wait=truekubectl create namespace curso-k8s3. Pod sueltokubectl run primer-pod -n curso-k8s --image=nginx:stable-alpine --port=80kubectl wait -n curso-k8s --for=condition=Ready pod/primer-pod --timeout=120skubectl get pods -n curso-k8s -o widekubectl describe pod primer-pod -n curso-k8skubectl logs primer-pod -n curso-k8skubectl exec primer-pod -n curso-k8s -- nginx -vkubectl delete pod primer-pod -n curso-k8skubectl get pods -n curso-k8sEl Pod no vuelve porque no tiene controlador.4. Deployment y réplicaskubectl create deployment web -n curso-k8s --image=nginx:stable-alpine --replicas=2kubectl rollout status deployment/web -n curso-k8s --timeout=120skubectl get deployments,replicasets,pods -n curso-k8s kubectl scale deployment/web -n curso-k8s --replicas=3kubectl rollout status deployment/web -n curso-k8s --timeout=120skubectl get pods -n curso-k8s -l app=web kubectl delete pod NOMBRE_DEL_POD -n curso-k8skubectl get pods -n curso-k8s -l app=web -w      # Ctrl+C para salirReto opcional: imagen rota y recuperaciónkubectl create deployment api -n curso-k8s --image=httpd:2.4-alpine --replicas=2kubectl rollout status deployment/api -n curso-k8s --timeout=120skubectl get pods -n curso-k8s -l app=webkubectl get pods -n curso-k8s -l app=apikubectl describe deployment api -n curso-k8s     # el contenedor se llama "httpd" kubectl set image deployment/api -n curso-k8s httpd=httpd:etiqueta-inventadakubectl get pods -n curso-k8s -l app=apikubectl describe pod -n curso-k8s -l app=api kubectl set image deployment/api -n curso-k8s httpd=httpd:2.4-alpinekubectl rollout status deployment/api -n curso-k8s --timeout=120skubectl rollout history deployment/api -n curso-k8skubectl get deployments -n curso-k8s5. Cierrekubectl delete namespace curso-k8s --wait=truekubectl get namespace curso-k8s                  # NotFound
Workshop 2: YAML en clase
Si existe un web previo:kubectl delete deployment web -n curso-k8s --ignore-not-found
namespace.yamlapiVersion: v1kind: Namespacemetadata:  name: curso-k8skubectl apply --dry-run=server -f namespace.yamlkubectl apply -f namespace.yamlkubectl get namespace curso-k8s
pod.yamlapiVersion: v1kind: Podmetadata:  name: primer-pod  namespace: curso-k8sspec:  containers:    - name: nginx      image: nginx:stable-alpine      ports:        - containerPort: 80
kubectl apply -f pod.yamlkubectl wait -n curso-k8s --for=condition=Ready pod/primer-pod --timeout=120skubectl get pods -n curso-k8s -o widekubectl delete -f pod.yamlkubectl get pods -n curso-k8s
web.yamlapiVersion: apps/v1kind: Deploymentmetadata:  name: web  namespace: curso-k8sspec:  replicas: 2  selector:    matchLabels:      app: web  template:    metadata:      labels:        app: web    spec:      containers:        - name: nginx          image: nginx:stable-alpine          ports:            - containerPort: 80
kubectl apply -f web.yamlkubectl rollout status deployment/web -n curso-k8s --timeout=120skubectl get deployments,replicasets,pods -n curso-k8sEditar el archivoCambia replicas: 3 en web.yaml y vuelve a aplicar:kubectl apply -f web.yamlkubectl rollout status deployment/web -n curso-k8s --timeout=120skubectl get pods -n curso-k8s -l app=webkubectl delete pod NOMBRE_DEL_POD -n curso-k8skubectl get pods -n curso-k8s -l app=web -w      # Ctrl+CCierrekubectl delete namespace curso-k8s --wait=truekubectl get namespace curso-k8s
Workshop 3: Práctica individual
1 y 2. Entorno y namespacekubectl version --clientkubectl config current-contextkubectl get nodeskubectl create namespace curso-k8skubectl get namespace curso-k8s
3. Crear tienda.yaml
apiVersion: apps/v1kind: Deploymentmetadata:  name: tienda  namespace: curso-k8sspec:  replicas: 2  selector:    matchLabels:      app: tienda  template:    metadata:      labels:        app: tienda    spec:      containers:        - name: nginx          image: nginx:stable-alpine          ports:            - containerPort: 80          resources:            requests:              cpu: 50m              memory: 32Mi            limits:              cpu: 200m              memory: 128Mi          readinessProbe:            httpGet:              path: /              port: 80            initialDelaySeconds: 2            periodSeconds: 5---apiVersion: v1kind: Servicemetadata:  name: tienda  namespace: curso-k8sspec:  selector:    app: tienda  ports:    - port: 80      targetPort: 80  type: ClusterIPkubectl apply --dry-run=server -f tienda.yaml4. Aplicar y comprobarkubectl apply -f tienda.yamlkubectl rollout status deployment/tienda -n curso-k8s --timeout=120skubectl get deployments,pods,services -n curso-k8skubectl get endpointslices -n curso-k8s -l kubernetes.io/service-name=tienda5. Probar HTTPkubectl run cliente-http -n curso-k8s --image=curlimages/curl:latest \  --restart=Never --rm -i --command -- \  curl --fail --retry 5 --retry-connrefused http://tienda6. Escalar editando el archivo (sin kubectl scale)sed -i 's/replicas: 2/replicas: 3/' tienda.yamlkubectl apply -f tienda.yamlkubectl get pods -n curso-k8s -l app=tienda7. Romper y arreglar labelscp tienda.yaml tienda-roto.yamlEn tienda-roto.yaml cambia solo las labels del template (template.metadata.labels) a app: tienda-mal. El selector.matchLabels se queda en app: tienda.kubectl apply --dry-run=server -f tienda-roto.yaml    # debe dar ERRORkubectl apply -f tienda.yamlkubectl get pods -n curso-k8s -l app=tienda8. Self-healingkubectl get pods -n curso-k8s -l app=tiendakubectl delete pod NOMBRE_DEL_POD -n curso-k8skubectl get pods -n curso-k8s -l app=tienda -w        # Ctrl+C9. Cierrekubectl delete namespace curso-k8s --wait=truekubectl get namespace curso-k8s                       # NotFound
